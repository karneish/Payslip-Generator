                  SHINECRAFT PAYSLIP - PgAdmin4 DATABASE QUERIES
================================================================================
Database : ShineCraft Payslip (PostgreSQL)
How to use: Open pgAdmin4 -> your ShineCraft database -> Query Tool (top-left
             lightning icon) -> paste any query below -> Run (F5)

NOTE ON TABLE NAMES:
  The tables are created by Prisma with exact model names (no @@map used).
  Because of this, table and column names preserve case (e.g. "Employee").
  In pgAdmin4, always write them with double quotes as shown below.

LIST OF TABLES:

  "GeneratedPayslip", "EmailLog", "UploadedFile", "ParsedSalaryData",
  "CompanyProfile", "SMTPConfig", "Setting", "AuditLog", "Department",

SELECT table_name

SECTION 3 : EMPLOYEES

-- 3.1 All employees (main details)
SELECT id, "employeeCode", "employeeName", email, "phoneNumber",
       department, designation, "panNumber", "aadharNumber",
       "joiningDate", "employmentStatus", category

-- 3.2 Employee with salary components

       "basicSalary", hra, da, "medicalAllowance", "travelAllowance",
       "specialAllowance", "otherAllowances", "grossSalary", "netSalary",
       "employmentStatus"
FROM "Employee"
ORDER BY "employeeName";

-- 3.3 Search employee by name or code
SELECT "employeeCode", "employeeName", email, department, designation,
       "employmentStatus"
FROM "Employee"
WHERE "employeeName" ILIKE '%Alpha%'
   OR "employeeCode" ILIKE '%EMP%'
ORDER BY "employeeName";

-- 3.4 Employees by department
SELECT department, COUNT(*) AS employee_count
FROM "Employee"
GROUP BY department
ORDER BY employee_count DESC;

-- 3.5 Active vs Inactive employees
SELECT "employmentStatus", COUNT(*) AS total
FROM "Employee"
GROUP BY "employmentStatus";

-- 3.6 Employees with their bank details
SELECT "employeeCode", "employeeName", "bankName", "bankAccountNumber", "ifscCode"
FROM "Employee"
ORDER BY "employeeName";

-- 3.7 Highest paid employees (by net salary)
SELECT "employeeCode", "employeeName", department, "netSalary"
FROM "Employee"

-- 3.8 Employees with Jibble integration details
SELECT "employeeCode", "employeeName", "jibbleEmployeeId", "jibbleUserId",
       "syncStatus", "lastSyncTime", "attendanceStatus"
FROM "Employee"

-- 3.9 Newly joined employees (this year)
SELECT "employeeCode", "employeeName", department, "joiningDate"
FROM "Employee"
WHERE EXTRACT(YEAR FROM "joiningDate") = EXTRACT(YEAR FROM CURRENT_DATE)
ORDER BY "joiningDate";

-- 3.10 Department-wise salary totals
SELECT department,
       COUNT(*) AS employees,
       SUM("grossSalary") AS total_gross,
       SUM("netSalary")   AS total_net
FROM "Employee"
GROUP BY department
ORDER BY total_net DESC;

SECTION 4 : ATTENDANCE
--------------------------------------------------------------------------------

-- 4.1 All attendance records
SELECT a.id, e."employeeCode", e."employeeName", a.date, a.status,
       a."clockInTime", a."clockOutTime", a."totalWorkedHours",
       a."overtimeHours", a."isHoliday", a."isWeekend", a."isLeave",
       a."leaveType", a.notes
FROM "Attendance" a
JOIN "Employee" e ON e.id = a."employeeId"
ORDER BY a.date DESC, e."employeeName";

-- 4.2 Attendance for a specific date
SELECT e."employeeCode", e."employeeName", a.status,
       a."totalWorkedHours", a."overtimeHours"
FROM "Attendance" a
JOIN "Employee" e ON e.id = a."employeeId"
WHERE a.date = '2026-07-15'::date
ORDER BY e."employeeName";

-- 4.3 Attendance for a specific month/year
SELECT e."employeeCode", e."employeeName",
       COUNT(*) FILTER (WHERE a.status = 'PRESENT') AS present,
       COUNT(*) FILTER (WHERE a.status = 'ABSENT')  AS absent,
       COUNT(*) FILTER (WHERE a.status = 'LEAVE')   AS on_leave,
       COUNT(*) FILTER (WHERE a.status = 'HOLIDAY') AS holidays
FROM "Attendance" a
JOIN "Employee" e ON e.id = a."employeeId"
WHERE EXTRACT(MONTH FROM a.date) = 7 AND EXTRACT(YEAR FROM a.date) = 2026
GROUP BY e."employeeCode", e."employeeName"
ORDER BY e."employeeName";

-- 4.4 Status-wise attendance count for a month
SELECT status, COUNT(*) AS total
FROM "Attendance"
WHERE EXTRACT(MONTH FROM date) = 7 AND EXTRACT(YEAR FROM date) = 2026
GROUP BY status;

-- 4.5 Employees with most absents in a month
SELECT e."employeeCode", e."employeeName",
       COUNT(*) FILTER (WHERE a.status = 'ABSENT') AS absent_days
FROM "Attendance" a
JOIN "Employee" e ON e.id = a."employeeId"
WHERE EXTRACT(MONTH FROM a.date) = 7 AND EXTRACT(YEAR FROM a.date) = 2026
GROUP BY e."employeeCode", e."employeeName"
HAVING COUNT(*) FILTER (WHERE a.status = 'ABSENT') > 0
ORDER BY absent_days DESC;

-- 4.6 Employees with overtime in a month
SELECT e."employeeCode", e."employeeName", a.date,
       a."overtimeHours", a."totalWorkedHours"
FROM "Attendance" a
JOIN "Employee" e ON e.id = a."employeeId"
WHERE EXTRACT(MONTH FROM a.date) = 7 AND EXTRACT(YEAR FROM a.date) = 2026
  AND a."overtimeHours" > 0
ORDER BY a."overtimeHours" DESC;

-- 4.7 Attendance for one employee (all history)
SELECT date, status, "clockInTime", "clockOutTime", "totalWorkedHours",
       "overtimeHours", notes
FROM "Attendance"
WHERE "employeeId" = 'PASTE_EMPLOYEE_UUID_HERE'
ORDER BY date DESC;

SECTION 6 : PAYSLIPS
--------------------------------------------------------------------------------

-- 6.1 All payslips with employee details
SELECT p.id, e."employeeCode", e."employeeName", p.month, p.year,
       p.status, p."grossSalary", p."netSalary", p."generatedAt",
       p."sentAt", p."createdAt"
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
ORDER BY p.year DESC, p.month DESC, e."employeeName";

-- 6.2 Payslips for a specific month/year
SELECT e."employeeCode", e."employeeName", p."basicSalary", p."totalEarnings",
       p."totalDeductions", p."grossSalary", p."netSalary", p.status
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
WHERE p.month = 7 AND p.year = 2026
ORDER BY e."employeeName";

-- 6.3 Payslips by status
SELECT p.status, COUNT(*) AS total
FROM "Payslip" p
GROUP BY p.status;

-- 6.4 Payslips filtered by status (e.g. GENERATED)
SELECT e."employeeCode", e."employeeName", p.month, p.year, p.status
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
WHERE p.status = 'GENERATED'
ORDER BY p.year DESC, p.month DESC;

-- 6.5 Full salary breakdown of one payslip
SELECT p.month, p.year,
       p."basicSalary", p.hra, p.da, p."medicalAllowance",
       p."travelAllowance", p."specialAllowance", p."otherAllowances",
       p.bonus, p.incentive, p."overtimePay", p."totalEarnings",
       p."pfDeduction", p."esiDeduction", p."professionalTax", p."incomeTax",
       p."leaveDeduction", p."lateDeduction", p."otherDeductions",
       p."advanceDeduction", p."totalDeductions",
       p."grossSalary", p."netSalary", p."amountInWords"
FROM "Payslip" p
WHERE p.id = 'PASTE_PAYSLIP_UUID_HERE';

-- 6.6 Payslip attendance summary
SELECT e."employeeCode", e."employeeName", p.month, p.year,
       p."totalDays", p."workingDays", p."presentDays", p."absentDays",
       p."leaveDays", p."holidayDays", p."weekendDays", p."lopDays",
       p."payableDays"
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
ORDER BY p.year DESC, p.month DESC;

-- 6.7 Highest and lowest net salary payslips
(SELECT e."employeeName", p.month, p.year, p."netSalary", 'HIGHEST' AS rank
 FROM "Payslip" p JOIN "Employee" e ON e.id = p."employeeId"
 ORDER BY p."netSalary" DESC LIMIT 1)
UNION ALL
(SELECT e."employeeName", p.month, p.year, p."netSalary", 'LOWEST' AS rank
 FROM "Payslip" p JOIN "Employee" e ON e.id = p."employeeId"
 ORDER BY p."netSalary" ASC LIMIT 1);

-- 6.8 Total payroll per month
SELECT p.month, p.year,
       COUNT(*) AS payslips,
       SUM(p."grossSalary") AS total_gross,
       SUM(p."netSalary")   AS total_net,
       SUM(p."totalDeductions") AS total_deductions
FROM "Payslip" p
GROUP BY p.year, p.month
ORDER BY p.year DESC, p.month DESC;

-- 6.9 Payslips for one employee across months
SELECT p.month, p.year, p."grossSalary", p."netSalary", p.status,
       p."generatedAt", p."sentAt"
FROM "Payslip" p
WHERE p."employeeId" = 'PASTE_EMPLOYEE_UUID_HERE'
ORDER BY p.year DESC, p.month DESC;

-- 6.10 Generated payslips (PDF) with download count
SELECT e."employeeCode", e."employeeName", gp.month, gp.year,
       gp."pdfPath", gp."pdfSize", gp."generatedAt",
       gp."downloadCount", gp."lastDownloadAt"
FROM "GeneratedPayslip" gp
JOIN "Employee" e ON e.id = gp."employeeId"
ORDER BY gp."generatedAt" DESC;

-- 6.11 Most downloaded payslips
SELECT e."employeeName", gp.month, gp.year, gp."downloadCount",
       gp."lastDownloadAt"
FROM "GeneratedPayslip" gp
JOIN "Employee" e ON e.id = gp."employeeId"
ORDER BY gp."downloadCount" DESC
LIMIT 10;

       pd.da, pd.bonus, pd.incentive, pd."netSalary", pd.workingDays,
       pd.leaves, pd.lop, pd."monthlySalary", pd.gross, pd.status,
       pd."isValid", pd."validationErrors"
FROM "ParsedSalaryData" pd
WHERE pd."uploadedFileId" = 'PASTE_UPLOAD_UUID_HERE'

       pd."netSalary", pd.status, pd."isValid", pd."validationErrors"
FROM "ParsedSalaryData" pd
JOIN "UploadedFile" u ON u.id = pd."uploadedFileId"
ORDER BY pd."createdAt" DESC;

-- 7.5 Invalid / failed parsed rows
SELECT u."fileName", pd."employeeName", pd."validationErrors"
FROM "ParsedSalaryData" pd
JOIN "UploadedFile" u ON u.id = pd."uploadedFileId"
WHERE pd."isValid" = false
   OR pd.status <> 'SAVED';

SECTION 14 : ADVANCED / REPORTING QUERIES
--------------------------------------------------------------------------------

-- 14.1 Month-wise payroll comparison
SELECT p.year, p.month,
       COUNT(*) AS employees_paid,
       ROUND(SUM(p."grossSalary")::numeric, 2) AS total_gross,
       ROUND(SUM(p."netSalary")::numeric, 2)   AS total_net,
       ROUND(AVG(p."netSalary")::numeric, 2)   AS average_net
FROM "Payslip" p
GROUP BY p.year, p.month
ORDER BY p.year, p.month;

-- 14.2 Department-wise payroll for a month
SELECT e.department,
       COUNT(*) AS employees,
       ROUND(SUM(p."netSalary")::numeric, 2) AS total_net_pay
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
WHERE p.month = 7 AND p.year = 2026
GROUP BY e.department
ORDER BY total_net_pay DESC;

-- 14.3 Employees without a payslip for a specific month
SELECT e."employeeCode", e."employeeName", e.department
FROM "Employee" e
WHERE e."employmentStatus" = 'Active'
  AND NOT EXISTS (
    SELECT 1 FROM "Payslip" p
    WHERE p."employeeId" = e.id AND p.month = 7 AND p.year = 2026
  )
ORDER BY e."employeeName";

-- 14.4 Employees with salary mismatch (employee table vs payslip)
SELECT e."employeeCode", e."employeeName",
       e."basicSalary" AS emp_basic, p."basicSalary" AS payslip_basic,
       e."netSalary"   AS emp_net,   p."netSalary"   AS payslip_net
FROM "Employee" e
JOIN "Payslip" p ON p."employeeId" = e.id AND p.month = 7 AND p.year = 2026
WHERE ABS(e."basicSalary" - p."basicSalary") > 0.01
   OR ABS(e."netSalary" - p."netSalary") > 0.01
ORDER BY e."employeeName";

-- 14.5 Full employee-payslip-attendance report for one month
SELECT e."employeeCode", e."employeeName", e.department, e.designation,
       s."presentDays", s."absentDays", s."lopDays", s."payableDays",
       p."grossSalary", p."totalDeductions", p."netSalary", p.status
FROM "Employee" e
LEFT JOIN "AttendanceSummary" s
       ON s."employeeId" = e.id AND s.month = 7 AND s.year = 2026
LEFT JOIN "Payslip" p
       ON p."employeeId" = e.id AND p.month = 7 AND p.year = 2026
ORDER BY e."employeeName";

-- 14.6 Data verification: upload vs saved payslips
SELECT u."fileName",
       u."recordCount" AS records_in_file,
       COUNT(DISTINCT pd."employeeCode") AS parsed_employees,
       COUNT(DISTINCT p."employeeId")    AS payslips_created
FROM "UploadedFile" u
LEFT JOIN "ParsedSalaryData" pd ON pd."uploadedFileId" = u.id
LEFT JOIN "Payslip" p ON p.year = u.year AND p.month = u.month
GROUP BY u.id, u."fileName", u."recordCount"
ORDER BY u."uploadedAt" DESC;

-- 14.7 Employees never synced with Jibble
SELECT "employeeCode", "employeeName", "syncStatus"
FROM "Employee"
WHERE "jibbleEmployeeId" IS NULL
   OR "syncStatus" = 'PENDING'
ORDER BY "employeeName";

SECTION 15 : HOUSEKEEPING / DIAGNOSTIC QUERIES (READ ONLY)
--------------------------------------------------------------------------------

-- 15.1 Duplicate employee codes (should return 0 rows)
SELECT "employeeCode", COUNT(*) AS duplicates
FROM "Employee"
GROUP BY "employeeCode"
HAVING COUNT(*) > 1;

-- 15.2 Duplicate aadhar numbers (should return 0 rows)
SELECT "aadharNumber", COUNT(*) AS duplicates
FROM "Employee"
GROUP BY "aadharNumber"
HAVING COUNT(*) > 1;

-- 15.3 Duplicate payslips (same employee + month + year) (should return 0 rows)
SELECT "employeeId", month, year, COUNT(*) AS duplicates
FROM "Payslip"
GROUP BY "employeeId", month, year
HAVING COUNT(*) > 1;

-- 15.4 Duplicate attendance (same employee + date) (should return 0 rows)
SELECT "employeeId", date, COUNT(*) AS duplicates
FROM "Attendance"
GROUP BY "employeeId", date
HAVING COUNT(*) > 1;

-- 15.5 Payslips with negative net salary (sanity check)
SELECT e."employeeCode", e."employeeName", p.month, p.year, p."netSalary"
FROM "Payslip" p
JOIN "Employee" e ON e.id = p."employeeId"
WHERE p."netSalary" < 0
ORDER BY p."netSalary";

-- 15.6 Employees with missing mandatory fields (sanity check)
SELECT "employeeCode", "employeeName",
       CASE WHEN email IS NULL OR email = '' THEN 'MISSING email' END,
       CASE WHEN "panNumber" IS NULL OR "panNumber" = '' THEN 'MISSING pan' END,
       CASE WHEN "aadharNumber" IS NULL OR "aadharNumber" = '' THEN 'MISSING aadhar' END,
       CASE WHEN "bankAccountNumber" IS NULL OR "bankAccountNumber" = '' THEN 'MISSING bank' END
FROM "Employee"
WHERE email = '' OR "panNumber" = '' OR "aadharNumber" = ''
   OR "bankAccountNumber" = ''
ORDER BY "employeeName";

-- 15.7 Orphan records check: attendance/payslip pointing to deleted employee
SELECT 'Attendance' AS orphan_type, COUNT(*) AS count
FROM "Attendance" a
LEFT JOIN "Employee" e ON e.id = a."employeeId"
WHERE e.id IS NULL
UNION ALL
SELECT 'Payslip', COUNT(*)
FROM "Payslip" p
LEFT JOIN "Employee" e ON e.id = p."employeeId"
WHERE e.id IS NULL;

-- 15.8 Recent activity across the whole system (last 7 days)
SELECT 'Employee created' AS event, "createdAt" AS occurred_at, "employeeName" AS detail
FROM "Employee" WHERE "createdAt" >= CURRENT_DATE - INTERVAL '7 days'
UNION ALL
SELECT 'Payslip generated', p."generatedAt", e."employeeName"
FROM "Payslip" p JOIN "Employee" e ON e.id = p."employeeId"
WHERE p."generatedAt" >= CURRENT_DATE - INTERVAL '7 days'
UNION ALL
SELECT 'Email sent', el."sentAt", el."recipientEmail"
FROM "EmailLog" el WHERE el."sentAt" >= CURRENT_DATE - INTERVAL '7 days'
UNION ALL
SELECT 'File uploaded', u."uploadedAt", u."fileName"
FROM "UploadedFile" u WHERE u."uploadedAt" >= CURRENT_DATE - INTERVAL '7 days'
ORDER BY occurred_at DESC;
